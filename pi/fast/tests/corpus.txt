import csv
from pathlib import Path


def load_rows(path):
    rows = []
    with Path(path).open(newline="", encoding="utf-8") as handle:
        for row in csv.DictReader(handle):
            rows.append(row)
    return rows


def sum_column(rows, column):
    total = 0.0
    for row in rows:
        value = row.get(column, "").strip()
        if not value:
            continue
        try:
            total += float(value)
        except ValueError:
            print(f"skipping bad value {value!r}")
    return total


def average_column(rows, column):
    total = sum_column(rows, column)
    count = sum(1 for row in rows if row.get(column, "").strip())
    if count == 0:
        return 0.0
    return total / count


def report(path, columns):
    rows = load_rows(path)
    lines = []
    for column in columns:
        total = sum_column(rows, column)
        mean = average_column(rows, column)
        lines.append(f"{column}: total={total:.2f} mean={mean:.2f}")
    return "\n".join(lines)


if __name__ == "__main__":
    print(report("sales.csv", ["amount", "tax", "discount"]))

The water cycle moves water between the sea, the air and the land. Heat from the sun turns water at the surface into vapour, which rises and cools until it forms clouds made of tiny droplets. When the droplets grow heavy they fall as rain or snow. Some of that water runs over the ground into streams and rivers, some soaks into the soil and feeds plants or slowly fills the rock layers below, and a part of it freezes in glaciers for many years. Rivers carry the water back to the sea, where the cycle begins again. People depend on every stage of it: farmers on rain at the right time, towns on rivers and wells, and power stations on water held behind dams. In German the same idea is short: Wasser verdunstet, bildet Wolken und fällt als Regen zurück auf die Erde. In Hebrew: המים מתאדים, יוצרים עננים ויורדים כגשם בחזרה אל האדמה.
