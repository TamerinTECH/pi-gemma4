cmake_minimum_required(VERSION 3.18)
project(cf_engine LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CF_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor-llama" CACHE PATH "Disposable pinned source overlay")
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_NUMBER 11398 CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMIT a7b94df CACHE STRING "" FORCE)
add_subdirectory(${CF_LLAMA_DIR} llama EXCLUDE_FROM_ALL)
target_sources(ggml-cpu PRIVATE adapter.cpp kernels.cpp hash.c)
target_include_directories(ggml-cpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CF_LLAMA_DIR}/vendor ${CF_LLAMA_DIR}/vendor/hash)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../tools/probe/probe.cpp" CF_PROBE_SOURCE)
string(REPLACE "            const float * lg = llama_get_logits_ith(ctx, -1);" "            const float * lg = llama_get_logits_ith(ctx, -1);\n            cf_record_logits(lg, n_vocab, ci, g);" CF_PROBE_SOURCE "${CF_PROBE_SOURCE}")
string(REPLACE "        const chunk_in &         cin  = chunks[ci];" "        cf_request_begin(ci);\n        const chunk_in &         cin  = chunks[ci];" CF_PROBE_SOURCE "${CF_PROBE_SOURCE}")
string(REPLACE "        const double tp2 = now_ms();" "        const double tp2 = now_ms();\n        cf_request_end(ci);" CF_PROBE_SOURCE "${CF_PROBE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cf-probe.cpp" "#include \"adapter.h\"\n${CF_PROBE_SOURCE}")
add_library(cf-probe OBJECT "${CMAKE_CURRENT_BINARY_DIR}/cf-probe.cpp")
target_include_directories(cf-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(cf-probe PRIVATE main=cf_probe_main)
target_link_libraries(cf-probe PRIVATE llama-common llama)
add_executable(cf-run main.cpp $<TARGET_OBJECTS:cf-probe>)
target_link_libraries(cf-run PRIVATE llama-common llama ggml-cpu)
add_executable(cf-kernel-test kernels_test.cpp kernels.cpp)
target_link_libraries(cf-kernel-test PRIVATE Threads::Threads)
set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/kernels.cpp"
    TARGET_DIRECTORY ggml-cpu cf-kernel-test
    PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
option(CF_EXPERIMENTAL_ASM "Build rejected-by-performance-gate assembly candidates for reproduction" OFF)
if(CF_EXPERIMENTAL_ASM AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    enable_language(ASM)
    foreach(CF_TARGET ggml-cpu cf-kernel-test)
        target_sources(${CF_TARGET} PRIVATE q4_tile4_a76.S)
        target_compile_definitions(${CF_TARGET} PRIVATE CF_ENABLE_ASM=1)
    endforeach()
endif()
option(CF_BUILD_COMPARISONS "Build isolated external kernel comparisons" OFF)
if(CF_BUILD_COMPARISONS)
    include(kleidi_sources.cmake)
    add_executable(cf-operator-bench operator_bench.cpp)
    target_include_directories(cf-operator-bench PRIVATE ${CF_LLAMA_DIR}/vendor ${CF_LLAMA_DIR}/ggml/src ${CF_LLAMA_DIR}/ggml/src/ggml-cpu)
    target_link_libraries(cf-operator-bench PRIVATE cf-kleidi-compare ggml ggml-cpu Threads::Threads)
endif()
